    <div id="diff">
      <button class="diff-btn selected" data-level="easy">쉬움</button>
      <button class="diff-btn" data-level="normal">보통</button>
      <button class="diff-btn" data-level="hard">어려움</button>
    </div>
    <p id="result-msg" style="color:#ffe066;font-size:1.1rem;margin-bottom:6px;"></p>

  <script>
    const TOTAL_TIME = 30;
    const HOLES = 9;

    // 난이도별 설정: [두더지 체류 시간(ms), 스폰 간격(ms)]
    const LEVELS = {
      easy:   { stay: 1400, interval: 1100 },
      normal: { stay: 900,  interval: 750  },
      hard:   { stay: 550,  interval: 480  },
    };

    let score = 0, combo = 0, timeLeft = TOTAL_TIME;
    let gameActive = false;
    let moleTimers = [];
    let spawnTimer = null;
    let countdownTimer = null;
    let selectedLevel = 'easy';

    // 그리드 생성
    const grid = document.getElementById('grid');
    const holes = [];

    for (let i = 0; i < HOLES; i++) {
      const hole = document.createElement('div');
      hole.className = 'hole';
      hole.innerHTML = `
        <div class="mole" id="mole${i}">
          <svg viewBox="0 0 70 80" xmlns="http://www.w3.org/2000/svg">
            <!-- 몸통 -->
            <ellipse cx="35" cy="55" rx="28" ry="26" fill="#8B5E3C"/>
            <!-- 배 -->
            <ellipse cx="35" cy="60" rx="17" ry="18" fill="#D4A876"/>
            <!-- 머리 -->
            <ellipse cx="35" cy="30" rx="26" ry="24" fill="#8B5E3C"/>
            <!-- 귀 왼 -->
            <ellipse cx="13" cy="14" rx="8" ry="10" fill="#8B5E3C"/>
            <ellipse cx="13" cy="14" rx="5" ry="7" fill="#c48a6e"/>
            <!-- 귀 오 -->
            <ellipse cx="57" cy="14" rx="8" ry="10" fill="#8B5E3C"/>
            <ellipse cx="57" cy="14" rx="5" ry="7" fill="#c48a6e"/>
            <!-- 눈 왼 -->
            <circle cx="24" cy="26" r="6" fill="#fff"/>
            <circle cx="25" cy="27" r="3.5" fill="#222"/>
            <circle cx="26" cy="25" r="1.2" fill="#fff"/>
            <!-- 눈 오 -->
            <circle cx="46" cy="26" r="6" fill="#fff"/>
            <circle cx="47" cy="27" r="3.5" fill="#222"/>
            <circle cx="48" cy="25" r="1.2" fill="#fff"/>
            <!-- 코 -->
            <ellipse cx="35" cy="36" rx="7" ry="5" fill="#c0478a"/>
            <!-- 콧구멍 -->
            <ellipse cx="32" cy="36" rx="2" ry="1.5" fill="#7a1a50"/>
            <ellipse cx="38" cy="36" rx="2" ry="1.5" fill="#7a1a50"/>
            <!-- 입 -->
            <path d="M28 42 Q35 48 42 42" stroke="#5a2d0c" stroke-width="2" fill="none" stroke-linecap="round"/>
            <!-- 수염 왼 -->
            <line x1="10" y1="38" x2="28" y2="40" stroke="#5a2d0c" stroke-width="1.5"/>
            <line x1="10" y1="42" x2="28" y2="42" stroke="#5a2d0c" stroke-width="1.5"/>
            <!-- 수염 오 -->
            <line x1="60" y1="38" x2="42" y2="40" stroke="#5a2d0c" stroke-width="1.5"/>
            <line x1="60" y1="42" x2="42" y2="42" stroke="#5a2d0c" stroke-width="1.5"/>
          </svg>
        </div>
        <div class="hole-inner"></div>
        <div class="ground"></div>
      `;
      hole.addEventListener('click', () => whack(i));
      grid.appendChild(hole);
      holes.push({
        el: hole,
        mole: hole.querySelector('.mole'),
        active: false,
      });
    }

    // 난이도 버튼
    document.querySelectorAll('.diff-btn').forEach(btn => {
      btn.addEventListener('click', () => {
        document.querySelectorAll('.diff-btn').forEach(b => b.classList.remove('selected'));
        btn.classList.add('selected');
        selectedLevel = btn.dataset.level;
      });
    });

    document.getElementById('startBtn').addEventListener('click', startGame);

    function startGame() {
      document.getElementById('overlay').style.display = 'none';
      score = 0; combo = 0; timeLeft = TOTAL_TIME;
      updateHUD();
      document.getElementById('timer-bar').style.width = '100%';
      document.getElementById('timer-bar').style.background = 'linear-gradient(90deg,#a8e063,#4caf50)';
      gameActive = true;
      spawnMoles();
      startCountdown();
    }

    function spawnMoles() {
      const cfg = LEVELS[selectedLevel];
      const active = holes.filter(h => h.active).length;
      // 최대 3개 동시 노출
      if (active < 3) {
        const available = holes.map((h,i) => h.active ? null : i).filter(i => i !== null);
        if (available.length > 0) {
          const idx = available[Math.floor(Math.random() * available.length)];
          showMole(idx, cfg.stay);
        }
      }
      if (gameActive) spawnTimer = setTimeout(spawnMoles, cfg.interval);
    }

    function showMole(idx, stay) {
      const h = holes[idx];
      if (h.active) return;
      h.active = true;
      h.mole.classList.add('up');
      moleTimers[idx] = setTimeout(() => hideMole(idx), stay);
    }

    function hideMole(idx) {
      const h = holes[idx];
      h.active = false;
      h.mole.classList.remove('up', 'whacked');
    }

    function whack(idx) {
      if (!gameActive) return;
      const h = holes[idx];
      if (!h.active || h.mole.classList.contains('whacked')) return;

      clearTimeout(moleTimers[idx]);
      h.mole.classList.add('whacked');
      setTimeout(() => hideMole(idx), 300);

      combo++;
      const pts = combo >= 5 ? 30 : combo >= 3 ? 20 : 10;
      score += pts;
      updateHUD();
      showPop(idx, pts, combo);
    }

    function showPop(idx, pts, combo) {
      const hole = holes[idx].el;
      const pop = document.createElement('div');
      pop.className = 'pop';
      pop.textContent = combo >= 3 ? `+${pts} 콤보!` : `+${pts}`;
      pop.style.left = '50%';
      pop.style.top = '10px';
      pop.style.transform = 'translateX(-50%)';
      hole.appendChild(pop);
      setTimeout(() => pop.remove(), 700);
    }

    function startCountdown() {
      countdownTimer = setInterval(() => {
        timeLeft--;
        const ratio = timeLeft / TOTAL_TIME;
        document.getElementById('timer-bar').style.width = (ratio * 100) + '%';
        if (ratio < 0.3)
          document.getElementById('timer-bar').style.background = 'linear-gradient(90deg,#ff6b6b,#e03131)';
        else if (ratio < 0.6)
          document.getElementById('timer-bar').style.background = 'linear-gradient(90deg,#ffe066,#f08c00)';
        updateHUD();
        if (timeLeft <= 0) endGame();
      }, 1000);
    }

    function endGame() {
      gameActive = false;
      clearTimeout(spawnTimer);
      clearInterval(countdownTimer);
      holes.forEach((h, i) => { clearTimeout(moleTimers[i]); hideMole(i); });

      const overlay = document.getElementById('overlay');
      document.getElementById('result-msg').textContent = '게임 종료! 최종 점수:';
      document.getElementById('final-score').textContent = score + '점';
      document.getElementById('final-score').style.display = 'block';
      document.getElementById('startBtn').textContent = '다시 하기';
      overlay.style.display = 'flex';
    }

    function updateHUD() {
      document.getElementById('score').textContent = score;
      document.getElementById('combo').textContent = combo;
      document.getElementById('timeLeft').textContent = timeLeft;
    }

    // 클릭 미스 시 콤보 리셋
    grid.addEventListener('click', e => {
      if (!gameActive) return;
      if (e.target.closest('.hole') && !e.target.closest('.mole')) {
        combo = 0;
        updateHUD();
      }
    });
  </script>
